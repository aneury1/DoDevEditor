        execute_process(COMMAND "${wxWidgets_CONFIG_EXECUTABLE}" --selected-config OUTPUT_VARIABLE DODEV_WX_CONFIG OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if(DODEV_WX_CONFIG MATCHES "gtk3")
            find_package(PkgConfig QUIET)
            if(PkgConfig_FOUND)
                pkg_check_modules(DODEV_GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
                if(TARGET PkgConfig::DODEV_GTK3)

                    target_compile_definitions(DoDevEditor PRIVATE DODEV_GTK3_THEME=1)
                    message(STATUS "DoDevEditor: GTK3 native theme CSS enabled")
                else()
                    message(STATUS "DoDevEditor: GTK3 development files absent; wxWidgets palette only")
                endif()
            endif()
        endif()
    endif()
    install(TARGETS DoDevEditor dodeveditor RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
    if(UNIX AND NOT APPLE)
        install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/dodev_install_desktop.cmake")

    add_library(hello_plugin MODULE plugins/examples/HelloPlugin/HelloPlugin.cpp)
    target_include_directories(hello_plugin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
    set_target_properties(hello_plugin PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugin-examples/hello"

    foreach(extension IN ITEMS ClangFormat Docker)
        string(TOLOWER "${extension}" extension_lower)
        add_library(${extension_lower}_plugin MODULE
            "plugins/examples/${extension}/${extension}Plugin.cpp")
        target_include_directories(${extension_lower}_plugin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
        set_target_properties(${extension_lower}_plugin PROPERTIES
            LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugin-examples/${extension_lower}"
            RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugin-examples/${extension_lower}")
        configure_file("${CMAKE_CURRENT_SOURCE_DIR}/plugins/examples/${extension}/plugin.json"
                       "${CMAKE_CURRENT_BINARY_DIR}/plugin-examples/${extension_lower}/plugin.json" COPYONLY)
    endforeach()
    # Docker is a bundled, opt-in native plugin. Retain the example output for SDK tests.
    add_custom_command(TARGET docker_plugin POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/plugins/docker"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:docker_plugin>" "${CMAKE_CURRENT_BINARY_DIR}/plugins/docker/$<TARGET_FILE_NAME:docker_plugin>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/plugins/examples/Docker/plugin.json" "${CMAKE_CURRENT_BINARY_DIR}/plugins/docker/plugin.json")
    install(TARGETS docker_plugin LIBRARY DESTINATION "${CMAKE_INSTALL_BINDIR}/plugins/docker" RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}/plugins/docker")
    install(FILES plugins/examples/Docker/plugin.json DESTINATION "${CMAKE_INSTALL_BINDIR}/plugins/docker")
    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/plugins/examples/HelloPlugin/plugin.json"
        "${CMAKE_CURRENT_BINARY_DIR}/plugin-examples/hello/plugin.json" COPYONLY)
endif()
