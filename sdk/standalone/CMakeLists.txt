cmake_minimum_required(VERSION 3.21)
project(DoDevEditorStandalonePlugins VERSION 5.2.4 LANGUAGES CXX)
include(GNUInstallDirs)
include(cmake/ProjectHelpers.cmake)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
# Portable process support is compiled into Git; no editor binary or editor core is linked.
add_library(dodev_terminal STATIC support/TerminalSession.cpp)
target_include_directories(dodev_terminal PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(dodev_terminal PUBLIC Threads::Threads)
if(UNIX)
    find_library(UTIL_LIBRARY util)
    if(UTIL_LIBRARY)
        target_link_libraries(dodev_terminal PUBLIC "${UTIL_LIBRARY}")
    endif()
endif()
add_subdirectory(thirdparty/GitHistory)
add_subdirectory(thirdparty/QRCode)
dodev_plugin_filename(dodev_qrcode_plugin)
add_subdirectory(plugins/examples/SimpleModule)
enable_testing()
add_executable(qrcode_payload_tests thirdparty/QRCode/tests/payload_tests.cpp)
target_link_libraries(qrcode_payload_tests PRIVATE dodev_qr_core)
dodev_test_assertions(qrcode_payload_tests)
add_test(NAME qrcode_binary_payload COMMAND qrcode_payload_tests)
if(UNIX AND NOT APPLE)
    add_executable(qrcode_plugin_smoke thirdparty/QRCode/tests/plugin_smoke.cpp)
    target_include_directories(qrcode_plugin_smoke PRIVATE "${PROJECT_SOURCE_DIR}/include")
    target_link_libraries(qrcode_plugin_smoke PRIVATE ${CMAKE_DL_LIBS})
    dodev_test_assertions(qrcode_plugin_smoke)
    add_test(NAME qrcode_plugin_smoke COMMAND qrcode_plugin_smoke "$<TARGET_FILE:dodev_qrcode_plugin>")
endif()
